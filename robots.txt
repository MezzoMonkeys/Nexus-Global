# access that wasn't already open — the point is that the intent is now explicit,
# so a later blanket Disallow can't silently cut off AI search visibility, which
# is the failure mode that is easy to introduce and very hard to notice.

Allow: /
Disallow: /ASSETS/
Disallow: /docs/

# AI answer engines and their training crawlers, allowed deliberately.
# Two different jobs are grouped here and it is worth keeping them straight:
#   - OAI-SearchBot, ChatGPT-User, PerplexityBot and Claude-User fetch pages to
#     answer a live question, and are the ones that produce a citation.
#   - GPTBot, ClaudeBot, CCBot and Google-Extended feed training and grounding
#     corpora. They earn no immediate citation but are how a model comes to know
